function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheetBudaya = ss.getSheetByName("Data_Budaya"); var rows = sheetBudaya.getDataRange().getValues(); var dataBudaya = []; for (var i = 1; i < rows.length; i++) { var row = rows[i]; dataBudaya.push({ "id": row[0], "nama": row[1], "kategori": row[2], "deskripsi": row[3], "lat": row[4], "lng": row[5], "gambar": row[6], "gambar360": row[7], "video": row[8], "gambarLama": row[9], "gambarBaru": row[10] }); } // Mengambil data statistik dari sheet "Statistik" var sheetStat = ss.getSheetByName("Statistik"); var statistik = { Situs: 0, Struktur: 0, Bangunan: 0, Benda: 0 }; if (sheetStat && sheetStat.getLastRow() >= 2) { statistik.Situs = Number(sheetStat.getRange(2, 1).getValue()) || 0; statistik.Struktur = Number(sheetStat.getRange(2, 2).getValue()) || 0; statistik.Bangunan = Number(sheetStat.getRange(2, 3).getValue()) || 0; statistik.Benda = Number(sheetStat.getRange(2, 4).getValue()) || 0; } var responseObj = { dataBudaya: dataBudaya, statistik: statistik }; return ContentService.createTextOutput(JSON.stringify(responseObj)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { var lock = LockService.getScriptLock(); lock.tryLock(10000); try { var spreadsheetId = "1G8h9I-8ZNhrY-kOliuf3daI8mSbeB1_eulZznsl8gMw"; var doc = SpreadsheetApp.openById(spreadsheetId); var data = JSON.parse(e.postData.contents); // Jika permintaan untuk memperbarui statistik if (data.action === "updateStat") { var sheetStat = doc.getSheetByName("Statistik"); if (!sheetStat) { sheetStat = doc.insertSheet("Statistik"); sheetStat.appendRow(["Situs", "Struktur", "Bangunan", "Benda"]); } if (sheetStat.getLastRow() < 2) { sheetStat.appendRow([0, 0, 0, 0]); } sheetStat.getRange(2, 1, 1, 4).setValues([[ data.statistik.Situs, data.statistik.Struktur, data.statistik.Bangunan, data.statistik.Benda ]]); return ContentService.createTextOutput(JSON.stringify({"result": "success", "type": "stat"})).setMimeType(ContentService.MimeType.JSON); } // Jika permintaan untuk input data budaya baru var sheetBudaya = doc.getSheetByName("Data_Budaya"); sheetBudaya.appendRow([ data.id, data.nama, data.kategori, data.deskripsi, data.lat, data.lng, data.gambar, data.gambar360, data.video, data.gambarLama, data.gambarBaru ]); return ContentService.createTextOutput(JSON.stringify({"result": "success", "type": "budaya"})).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({"result": "error", "error": err.toString()})).setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } }
